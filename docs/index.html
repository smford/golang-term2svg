golang-term2svg

Convert live terminal commands, ANSI streams, and screen models into crisp vector SVG graphics in pure Go.

Go Module Installation

go get github.com/smford/golang-term2svg@v0.1.0

CLI Installation

brew install smford/tap/term2svg

Or download precompiled standalone binaries from GitHub Releases.

Quick Example

package main

import (
    "context"
    "os"
    "github.com/smford/golang-term2svg"
)

func main() {
    svg, err := term2svg.CommandToSVG(context.Background(), "kubectl", []string{"get", "pods"}, term2svg.Options{
        Title:       "Kubernetes Pods",
        Theme:       "catppuccin",
        AutoCrop:    true,
        ShowCursor:  true,
    })
    if err != nil {
        panic(err)
    }

    _ = os.WriteFile("pods.svg", []byte(svg), 0644)
}
